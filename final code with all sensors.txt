import cv2
import serial
import time
import requests
import numpy as np
from ultralytics import YOLO
from datetime import datetime

# ------------------ CONFIG ------------------
ESP32_CAM_URL = "http://10.10.32.245/capture"
SERIAL_PORT = "COM3"

BOT_TOKEN = "YOUR_TOKEN"
CHAT_ID = "YOUR_CHAT_ID"

LATITUDE = 11.9416
LONGITUDE = 79.8083

ANIMALS = ["cow","dog","horse","sheep","elephant","bear","bird","pig"]

# ------------------ TELEGRAM ------------------
def send_telegram_image(image_path, caption):
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendPhoto"
    with open(image_path, "rb") as photo:
        requests.post(url, files={"photo": photo}, data={"chat_id": CHAT_ID, "caption": caption})

# ------------------ START ------------------
print("Starting program...")

ser = serial.Serial(SERIAL_PORT, 115200, timeout=1)
time.sleep(2)

model = YOLO("yolov8n.pt")

last_state = '0'
last_sent_time = 0
cooldown = 30  # seconds

while True:
    try:
        # 📸 Get image from ESP32-CAM
        response = requests.get(ESP32_CAM_URL, timeout=5)
        img_array = np.array(bytearray(response.content), dtype=np.uint8)
        frame = cv2.imdecode(img_array, -1)

        if frame is None:
            continue

        # 🤖 YOLO Detection
        results = model(frame, verbose=False)

        detected = False
        detected_animal = None

        for r in results:
            for box in r.boxes:
                label = model.names[int(box.cls[0])]

                if label in ANIMALS:
                    detected = True
                    detected_animal = label

                    # 📦 Bounding box
                    x1, y1, x2, y2 = map(int, box.xyxy[0])

                    # 🟩 Draw rectangle
                    cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 2)

                    # 🏷️ Label
                    cv2.putText(frame, label, (x1, y1 - 10),
                                cv2.FONT_HERSHEY_SIMPLEX, 0.9, (0, 255, 0), 2)

                    print("Animal detected:", label)
                    break

        # 📏 Distance from Arduino
        distance = None
        if detected:
            ser.write(b'D')
            time.sleep(0.2)

            while ser.in_waiting:
                line = ser.readline().decode(errors='ignore').strip()
                if line.startswith("DIST:"):
                    distance = float(line.replace("DIST:", ""))
                    print("Distance:", distance)

        # 🔊 BUZZER LOGIC (FIXED)
        if detected and distance is not None and 0 < distance < 300:
            if last_state != '1':
                ser.write(b'1')
                print("Buzzer ON")
                last_state = '1'
        else:
            if last_state != '0':
                ser.write(b'0')
                print("Buzzer OFF")
                last_state = '0'

        # 📱 TELEGRAM ALERT
        if detected and detected_animal and distance and distance > 0:
            if time.time() - last_sent_time > cooldown:

                last_sent_time = time.time()

                image_name = "detected.jpg"
                cv2.imwrite(image_name, frame)

                current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

                message = f"""🚨 Animal Detected!
Type: {detected_animal}
Distance: {distance:.2f} cm

📍 Location:
https://maps.google.com/?q={LATITUDE},{LONGITUDE}

🕒 Time: {current_time}
"""

                send_telegram_image(image_name, message)
                print("Telegram sent")

        # 🖥️ Display
        cv2.imshow("Detection", frame)

        if cv2.waitKey(1) == 27:
            break

    except Exception as e:
        print("Error:", e)
        time.sleep(2)

cv2.destroyAllWindows()
ser.close()
